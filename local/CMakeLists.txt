cmake_minimum_required(VERSION 3.16)
find_package(ASC REQUIRED)
project(local_batch_matmul_max_sum LANGUAGES ASC CXX)
set(CMAKE_CXX_STANDARD 14)
if(NOT DEFINED NPU_ARCH)
    set(NPU_ARCH "dav-2201")
endif()
add_executable(baseline_runner runner.asc)
target_include_directories(baseline_runner PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(baseline_runner PRIVATE tiling_api register platform unified_dlog dl m graph_base)
target_compile_options(baseline_runner PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${NPU_ARCH}>)
